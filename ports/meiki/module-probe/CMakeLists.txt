cmake_minimum_required(VERSION 3.16)
set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
project(meiki_module_probe C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(MEIKI_MODULE "" CACHE FILEPATH "Built isolated meiki-engine.suprx")
set(MEIKI_BUNDLE_DIR "" CACHE PATH "Private four-crop bundle with reference.raw")
if(NOT EXISTS "${MEIKI_MODULE}" OR NOT EXISTS "${MEIKI_BUNDLE_DIR}/reference.raw")
  message(FATAL_ERROR "Set MEIKI_MODULE and MEIKI_BUNDLE_DIR")
endif()
set(SCREEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../vita/app")
add_executable(meiki_module_probe main.c ../decode.c ../preprocess.c ../detect_preprocess.c "${SCREEN_DIR}/screen.c")
target_include_directories(meiki_module_probe PRIVATE .. "${SCREEN_DIR}")
target_compile_features(meiki_module_probe PRIVATE c_std_99)
target_compile_options(meiki_module_probe PRIVATE -O2 -ffp-contract=off -Wall -Wextra -ffunction-sections -fdata-sections)
target_link_options(meiki_module_probe PRIVATE -Wl,--gc-sections)
target_link_libraries(meiki_module_probe m SceSysmem_stub SceKernelThreadMgr_stub SceIofilemgr_stub SceLibKernel_stub SceProcessmgr_stub SceKernelModulemgr_stub ScePower_stub SceDisplay_stub SceCtrl_stub)
vita_create_self(eboot.bin meiki_module_probe)
set(BUNDLE_FILES FILE "${MEIKI_MODULE}" "data/meiki-engine.suprx")
file(GLOB DATA_FILES "${MEIKI_BUNDLE_DIR}/*.f32" "${MEIKI_BUNDLE_DIR}/*.rgba")
foreach(DATA_FILE IN LISTS DATA_FILES)
  get_filename_component(DATA_NAME "${DATA_FILE}" NAME)
  list(APPEND BUNDLE_FILES FILE "${DATA_FILE}" "data/${DATA_NAME}")
endforeach()
foreach(DATA_NAME cases.tsv meiki-stream-int8.mnn font.bin font-OFL.txt reference.raw)
  list(APPEND BUNDLE_FILES FILE "${MEIKI_BUNDLE_DIR}/${DATA_NAME}" "data/${DATA_NAME}")
endforeach()
foreach(DATA_NAME detect.tsv meiki-detect-int8.mnn meiki-detect-oom.mnn)
  if(EXISTS "${MEIKI_BUNDLE_DIR}/${DATA_NAME}")
    list(APPEND BUNDLE_FILES FILE "${MEIKI_BUNDLE_DIR}/${DATA_NAME}" "data/${DATA_NAME}")
  endif()
endforeach()
vita_create_vpk(meiki-module-probe.vpk MKBRC0001 eboot.bin NAME "Meiki module probe" VERSION "00.01" ${BUNDLE_FILES})
